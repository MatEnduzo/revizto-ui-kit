## 🎯 IMPLEMENTATION RULES

### Rule 1: Import First
ALWAYS start with import statement:

```jsx
import { 
  Button, 
  Input, 
  Modal,
  // ... other needed components
} from './revizto-ui-tailwind';
```

### Rule 2: Use Only Documented Props
DO NOT add custom props. If a prop isn't listed above, DON'T USE IT.

```jsx
// ✅ CORRECT
<Button variant="primary" onClick={handleClick}>
  Save
</Button>

// ❌ WRONG - 'color' prop doesn't exist
<Button color="blue" onClick={handleClick}>
  Save
</Button>
```

### Rule 3: Extend with className Only
To customize appearance, use `className` prop with Tailwind:

```jsx
// ✅ CORRECT
<Button 
  variant="primary"
  className="w-full mt-4"
>
  Full Width Button
</Button>

// ❌ WRONG - creating custom styled button
const CustomButton = styled(Button)`
  width: 100%;
`;
```

### Rule 4: State Management
Use React hooks for state, not custom state components:

```jsx
// ✅ CORRECT
import { useState } from 'react';
import { Input } from './revizto-ui-tailwind';

function Form() {
  const [value, setValue] = useState('');
  
  return (
    <Input 
      value={value}
      onChange={(e) => setValue(e.target.value)}
    />
  );
}

// ❌ WRONG - creating state wrapper
function StatefulInput({ ... }) {
  // Custom implementation
}
```

## 🚫 ANTI-PATTERNS TO AVOID

### Anti-pattern 1: Creating Wrappers
```jsx
// ❌ NEVER DO THIS
function MyButton({ children, ...props }) {
  return (
    <button className="custom-button" {...props}>
      {children}
    </button>
  );
}

// ✅ DO THIS INSTEAD
import { Button } from './revizto-ui-tailwind';
<Button variant="primary">{children}</Button>
```

### Anti-pattern 2: Style Overrides
```jsx
// ❌ NEVER DO THIS
<Button 
  variant="primary"
  style={{ backgroundColor: 'red' }}
>
  Button
</Button>

// ✅ DO THIS INSTEAD
<Button 
  variant="primary"
  className="!bg-red-600"
>
  Button
</Button>
```

### Anti-pattern 3: Component Recreation
```jsx
// ❌ NEVER DO THIS
function CustomInput({ label, ...props }) {
  return (
    <div>
      <label>{label}</label>
      <input {...props} />
    </div>
  );
}

// ✅ DO THIS INSTEAD
import { Input } from './revizto-ui-tailwind';

function FormField({ label, ...props }) {
  return (
    <div>
      {label && <label className="block mb-2">{label}</label>}
      <Input {...props} />
    </div>
  );
}
```

```jsx
I'll create a login form using Revizto UI components:

import { Input, Button } from './revizto-ui-tailwind';
import { useState } from 'react';

function LoginForm() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [errors, setErrors] = useState({});

  const handleSubmit = () => {
    const newErrors = {};
    if (!email) newErrors.email = 'Email required';
    if (!password) newErrors.password = 'Password required';
    
    if (Object.keys(newErrors).length > 0) {
      setErrors(newErrors);
      return;
    }
    
    console.log('Login:', { email, password });
  };

  return (
    <div className="max-w-md mx-auto p-6 space-y-4">
      <h2 className="text-2xl font-bold mb-6">Login</h2>
      
      <Input
        placeholder="Email"
        value={email}
        onChange={(e) => {
          setEmail(e.target.value);
          setErrors({ ...errors, email: '' });
        }}
        error={errors.email}
      />
      
      <Input
        placeholder="Password"
        value={password}
        onChange={(e) => {
          setPassword(e.target.value);
          setErrors({ ...errors, password: '' });
        }}
        error={errors.password}
      />
      
      <Button 
        variant="primary" 
        onClick={handleSubmit}
        className="w-full"
      >
        Sign In
      </Button>
    </div>
  );
}
```

- Input: For email and password fields with error handling
- Button: For form submission
